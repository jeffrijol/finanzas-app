Paso 2: Keyboard Shortcuts (Prioridad Alta)

// En HomePage.tsx
useEffect(() => {
  const handleKeyDown = (e: KeyboardEvent) => {
    // Ctrl+S / Cmd+S para guardar
    if ((e.ctrlKey || e.metaKey) && e.key === 's') {
      e.preventDefault();
      handleSavePartial();
      return;
    }
    
    // Ctrl+Enter / Cmd+Enter para finalizar
    if ((e.ctrlKey || e.metaKey) && e.key === 'Enter') {
      e.preventDefault();
      if (canFinalize) handleFinalize();
      return;
    }
    
    // Escape para cancelar selección múltiple
    if (e.key === 'Escape' && bulkMode) {
      setBulkMode(false);
    }
  };
  
  window.addEventListener('keydown', handleKeyDown);
  return () => window.removeEventListener('keydown', handleKeyDown);
}, [handleSavePartial, handleFinalize, canFinalize, bulkMode]);

Paso 4: Batch Operations IMPORTANTE AQUI PRESTAR ATENCION

ADVERTENCIA SOBRE BATCH OPERATIONS

Problema potencial: Tu plan dice "re-introduces the bulk assignment logic you previously cancelled". Si antes decidiste NO hacer bulk operations, reconsideremos si realmente las necesitas.
Análisis Costo/Beneficio:

COSTOS:

    Complejidad UI adicional (checkboxes, estados de selección)

    Nuevos edge cases (¿qué pasa si seleccionas draft + synced?)

    Posible confusión para el usuario (dos formas de asignar)

    Más código para mantener

BENEFICIOS:

    Ahorro de clicks para asignar mismo item a muchas transacciones

    Flexibilidad para operaciones masivas

// Implementación minimalista
const useBatchOperations = () => {
  const [selectedIds, setSelectedIds] = useState<string[]>([]);
  const [isBatchMode, setIsBatchMode] = useState(false);
  
  const toggleBatchMode = () => {
    setIsBatchMode(!isBatchMode);
    if (isBatchMode) setSelectedIds([]); // Limpiar al salir
  };
  
  const handleBulkAssign = (itemId: string) => {
    // Aplicar a todas seleccionadas
    selectedIds.forEach(id => {
      updateTransactionItem(id, itemId);
    });
    
    // Feedback y auto-exit
    toast({
      title: `✅ Asignado a ${selectedIds.length} transacciones`,
      description: "Item aplicado correctamente",
    });
    
    // Auto-exit después de 2 segundos
    setTimeout(() => {
      setIsBatchMode(false);
      setSelectedIds([]);
    }, 2000);
  };
  
  return {
    isBatchMode,
    selectedIds,
    toggleBatchMode,
    handleBulkAssign,
    setSelectedIds,
  };
};
