// packages/frontend/src/hooks/useDashboardContext.ts
import { useMemo } from 'react';
import { useFilters } from './useFilters';

export type DashboardLevel = 
  | 'general'        // Sin filtros
  | 'year'           // Solo año seleccionado
  | 'quarter'        // Año + trimestre
  | 'type'           // Tipo de item seleccionado
  | 'item'           // Item específico
  | 'category';      // Categoría específica

export const useDashboardContext = () => {
  const filters = useFilters();
  
  const level = useMemo<DashboardLevel>(() => {
    if (filters.itemId) return 'item';
    if (filters.category) return 'category';
    if (filters.itemType) return 'type';
    if (filters.quarter) return 'quarter';
    if (filters.year) return 'year';
    return 'general';
  }, [filters]);
  
  const title = useMemo(() => {
    switch (level) {
      case 'item': return `Análisis: ${filters.itemName}`;
      case 'category': return `Categoría: ${filters.category}`;
      case 'type': return `Tipo: ${filters.itemType}`;
      case 'quarter': return `Trimestre ${filters.quarter} ${filters.year}`;
      case 'year': return `Año ${filters.year}`;
      default: return 'Dashboard General';
    }
  }, [level, filters]);
  
  return {
    level,
    title,
    filters,
    description: getDescription(level, filters),
    availableCharts: getAvailableCharts(level),
  };
};

// Función helper para determinar gráficos disponibles
const getAvailableCharts = (level: DashboardLevel): string[] => {
  const charts = {
    general: ['annual-summary', 'category-distribution', 'monthly-trend', 'top-items'],
    year: ['quarterly-breakdown', 'type-comparison', 'month-over-month', 'category-ranking'],
    quarter: ['monthly-detail', 'category-breakdown', 'year-over-year', 'quarter-items'],
    type: ['item-performance', 'internal-distribution', 'type-trend', 'vs-average'],
    item: ['category-breakdown', 'item-timeline', 'vs-type-average', 'kpi-cards'],
    category: ['item-distribution', 'category-timeline', 'vs-other-categories', 'subcategory-breakdown'],
  };
  return charts[level];
};

export const TypeItemsComparisonChart = () => {
  const { filters } = useDashboardContext();
  const { data, isLoading } = useQuery({
    queryKey: ['type-items', filters.itemType],
    queryFn: () => apiClient.getTypeItemsComparison(filters.itemType!),
    enabled: !!filters.itemType,
  });
  
  if (isLoading) return <LoadingSpinner />;
  if (!data) return <NoData />;
  
  return (
    <BarChart
      data={{
        labels: data.items.map(item => item.nombre),
        datasets: [
          {
            label: 'Ingresos',
            data: data.items.map(item => item.ingresos),
            backgroundColor: '#10B981',
          },

            label: 'Gastos',

            display: true,

// packages/backend/src/controllers/analytics.controller.ts
export class AnalyticsController {
  static async getTypeStats(req: Request, res: Response) {
    const { typeId } = req.params;
    const { startDate, endDate } = req.query;
    
    // 1. Obtener todos los items de este tipo
    const items = await prisma.item.findMany({
      where: {
        // Aquí necesitas una relación tipo-item (si no la tienes, hay que crearla)
        // O puedes usar metadata o un campo tipo en Item
      },
    });
    
    // 2. Para cada item, obtener sus transacciones en el período
    const itemsWithStats = await Promise.all(
      items.map(async (item) => {
        const transactions = await prisma.transaction.findMany({
          where: {
            itemAsignadoId: item.id,
            fechaValor: {
              gte: startDate ? new Date(startDate) : undefined,
              lte: endDate ? new Date(endDate) : undefined,
            },
          },
        });
        
        const ingresos = transactions.filter(t => t.importe > 0)
          .reduce((sum, t) => sum + t.importe, 0);
        const gastos = transactions.filter(t => t.importe < 0)
          .reduce((sum, t) => sum + Math.abs(t.importe), 0);
        
        return {
          itemId: item.id,
          nombre: item.nombre,
          ingresos,
          gastos,
          neto: ingresos - gastos,
          transactionCount: transactions.length,
        };
      })
    );
    
    // 3. Obtener distribución por categoría dentro de este tipo
    const categoryDistribution = await prisma.transaction.groupBy({
      by: ['categoria'],
      where: {
        itemAsignadoId: { in: items.map(i => i.id) },
        fechaValor: {
          gte: startDate ? new Date(startDate) : undefined,
          lte: endDate ? new Date(endDate) : undefined,
        },
      },
      _sum: { importe: true },
      _count: { id: true },
    });
    
    res.json({
      type: typeId,
      period: { startDate, endDate },
      items: itemsWithStats,
      categoryDistribution: categoryDistribution.map(cat => ({
        categoria: cat.categoria,
        total: cat._sum.importe,
        count: cat._count.id,
        tipo: cat._sum.importe > 0 ? 'ingreso' : 'gasto',
      })),
      summary: {
        totalIngresos: itemsWithStats.reduce((sum, item) => sum + item.ingresos, 0),
        totalGastos: itemsWithStats.reduce((sum, item) => sum + item.gastos, 0),
        totalNeto: itemsWithStats.reduce((sum, item) => sum + item.neto, 0),
        totalItems: items.length,
      },
    });
  }
  
  // Métodos similares para item, category, etc.
}
