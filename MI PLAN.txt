📋 Plan de Implementación: 
Objetivo

Generar PDFs con gráficos vectoriales idénticos a los del dashboard, usando los mismos componentes Recharts y estilos.
Estrategia Técnica

Renderizar los componentes Recharts en un entorno controlado (virtual DOM con JSDOM), capturar el SVG resultante y embeberlo en el PDF generado con @react-pdf/renderer.

2.1 Renderizador de Gráficos Recharts a SVG (chart-renderer.ts)

import { JSDOM } from 'jsdom';
import React from 'react';
import { renderToString } from 'react-dom/server';
import {
  BarChart,
  Bar,
  PieChart,
  Pie,
  Cell,
  XAxis,
  YAxis,
  CartesianGrid,
  Tooltip,
  Legend,
  ResponsiveContainer,
  LineChart,
  Line
} from 'recharts';

// Configurar DOM virtual para React
const setupVirtualDOM = () => {
  const dom = new JSDOM('<!DOCTYPE html><html><body></body></html>');
  global.window = dom.window as any;
  global.document = dom.window.document;
  global.navigator = dom.window.navigator;
  global.HTMLElement = dom.window.HTMLElement;
  global.SVGElement = dom.window.SVGElement;
};

/**
 * Renderiza un gráfico de barras a SVG string
 */
export const renderBarChartToSVG = (
  data: Array<{ name: string; ingresos: number; gastos: number }>,
  dimensions: { width: number; height: number } = { width: 600, height: 300 }
): string => {
  setupVirtualDOM();
  
  const chart = (
    <div style={{ width: dimensions.width, height: dimensions.height }}>
      <ResponsiveContainer width="100%" height="100%">
        <BarChart data={data} margin={{ top: 20, right: 30, left: 20, bottom: 5 }}>
          <CartesianGrid strokeDasharray="3 3" stroke="#e5e7eb" />
          <XAxis 
            dataKey="name" 
            stroke="#6b7280"
            fontSize={12}
            tickLine={false}
          />
          <YAxis 
            stroke="#6b7280"
            fontSize={12}
            tickLine={false}
            tickFormatter={(value) => `€${value.toLocaleString()}`}
          />
          <Tooltip 
            formatter={(value) => [`€${Number(value).toLocaleString()}`, '']}
            contentStyle={{
              backgroundColor: '#1f2937',
              border: 'none',
              borderRadius: '6px',
              color: 'white'
            }}
          />
          <Legend />
          <Bar 
            dataKey="ingresos" 
            name="Ingresos" 
            fill="#10b981" 
            radius={[4, 4, 0, 0]}
            maxBarSize={40}
          />
          <Bar 
            dataKey="gastos" 
            name="Gastos" 
            fill="#ef4444" 
            radius={[4, 4, 0, 0]}
            maxBarSize={40}
          />
        </BarChart>
      </ResponsiveContainer>
    </div>
  );
  
  return renderToString(chart);
};

/**
 * Renderiza un gráfico de torta a SVG string
 */
export const renderPieChartToSVG = (
  data: Array<{ name: string; value: number }>,
  colors: string[] = ['#3b82f6', '#10b981', '#8b5cf6', '#f59e0b', '#ef4444', '#06b6d4'],
  dimensions: { width: number; height: number } = { width: 400, height: 300 }
): string => {
  setupVirtualDOM();
  
  const chart = (
    <div style={{ width: dimensions.width, height: dimensions.height }}>
      <ResponsiveContainer width="100%" height="100%">
        <PieChart>
          <Pie
            data={data}
            cx="50%"
            cy="50%"
            labelLine={false}
            label={({ name, percent }) => `${name}: ${(percent * 100).toFixed(0)}%`}
            outerRadius={100}
            fill="#8884d8"
            dataKey="value"
          >
            {data.map((entry, index) => (
              <Cell key={`cell-${index}`} fill={colors[index % colors.length]} />
            ))}
          </Pie>
          <Tooltip 
            formatter={(value) => `€${Number(value).toLocaleString()}`}
            contentStyle={{
              backgroundColor: '#1f2937',
              border: 'none',
              borderRadius: '6px',
              color: 'white'
            }}
          />
          <Legend />
        </PieChart>
      </ResponsiveContainer>
    </div>
  );
  
  return renderToString(chart);
};

/**
 * Extrae solo el SVG del string renderizado
 */
export const extractSVGFromRender = (htmlString: string): string => {
  const match = htmlString.match(/<svg[\s\S]*?<\/svg>/);
  return match ? match[0] : '';
};

2.2 Hook Principal de Exportación (use-pdf-export.ts)

import { useCallback, useState } from 'react';
import { useQueryClient } from '@tanstack/react-query';
import { useDashboardStore } from '@/stores/dashboard-store';
import { useToast } from '@/components/ui/use-toast';
import { 
  renderBarChartToSVG, 
  renderPieChartToSVG,
  extractSVGFromRender 
} from '@/lib/pdf-generator/chart-renderer';
import { buildDashboardPDF } from '@/lib/pdf-generator/pdf-builder';
import { saveAs } from 'file-saver';

export const usePDFExport = () => {
  const [isGenerating, setIsGenerating] = useState(false);
  const { toast } = useToast();
  const queryClient = useQueryClient();
  const { filters } = useDashboardStore();

  const generateDashboardPDF = useCallback(async () => {
    setIsGenerating(true);
    
    try {
      // 1. Obtener datos actuales del cache de React Query
      const queryData = queryClient.getQueryData<any>(['dashboard-stats', filters]);
      
      if (!queryData) {
        throw new Error('No hay datos disponibles para generar el reporte');
      }

      // 2. Preparar datos para los gráficos (basado en tu estructura actual)
      const monthlyData = queryData.monthlyTrend || [];
      const expensesByCategory = queryData.expensesByCategory || [];
      const incomeByCategory = queryData.incomeByCategory || [];
      const transactions = queryData.transactions || [];

      // 3. Renderizar gráficos a SVG
      const barChartSVG = extractSVGFromRender(
        renderBarChartToSVG(monthlyData, { width: 700, height: 350 })
      );
      
      const expensesPieSVG = extractSVGFromRender(
        renderPieChartToSVG(
          expensesByCategory,
          ['#ef4444', '#f87171', '#fca5a5', '#fecaca', '#fee2e2'],
          { width: 350, height: 300 }
        )
      );
      
      const incomePieSVG = extractSVGFromRender(
        renderPieChartToSVG(
          incomeByCategory,
          ['#10b981', '#34d399', '#6ee7b7', '#a7f3d0', '#d1fae5'],
          { width: 350, height: 300 }
        )
      );

      // 4. Construir el objeto de datos para el PDF
      const reportData = {
        metadata: {
          title: `Reporte Financiero ${filters.year} ${filters.quarter === 'all' ? 'Anual' : `Q${filters.quarter}`}`,
          generatedAt: new Date().toLocaleDateString('es-ES', {
            weekday: 'long',
            year: 'numeric',
            month: 'long',
            day: 'numeric',
            hour: '2-digit',
            minute: '2-digit'
          }),
          filters: {
            ...filters,
            periodLabel: filters.quarter === 'all' 
              ? `Año completo ${filters.year}`
              : `Trimestre ${filters.quarter} de ${filters.year}`
          }
        },
        summaries: {
          totalTransactions: transactions.length,
          totalIncome: monthlyData.reduce((sum, item) => sum + (item.ingresos || 0), 0),
          totalExpenses: monthlyData.reduce((sum, item) => sum + (item.gastos || 0), 0),
          netBalance: monthlyData.reduce((sum, item) => sum + (item.ingresos || 0) - (item.gastos || 0), 0)
        },
        charts: {
          barChartSVG,
          expensesPieSVG,
          incomePieSVG
        },
        sampleTransactions: transactions.slice(0, 15) // Mostrar solo 15 en el PDF
      };

      // 5. Generar y descargar el PDF
      const pdfBlob = await buildDashboardPDF(reportData);
      const filename = `reporte-financiero-${filters.year}-${
        filters.quarter === 'all' ? 'anual' : `q${filters.quarter}`
      }-${Date.now()}.pdf`;
      
      saveAs(pdfBlob, filename);

      // 6. Notificación de éxito
      toast({
        title: '✅ Reporte generado',
        description: `El PDF se ha descargado como "${filename}"`,
        duration: 5000,
      });

      return true;
    } catch (error) {
      console.error('Error generando PDF:', error);
      
      toast({
        title: '❌ Error al generar reporte',
        description: error instanceof Error ? error.message : 'Intenta nuevamente',
        variant: 'destructive',
        duration: 5000,
      });
      
      return false;
    } finally {
      setIsGenerating(false);
    }
  }, [queryClient, filters, toast]);

  return {
    generateDashboardPDF,
    isGenerating
  };
};

/**
 * Función principal que genera el PDF como Blob
 */
export const buildDashboardPDF = async (data: any): Promise<Blob> => {
  const pdfElement = <DashboardPDF data={data} />;
  const blob = await pdf(pdfElement).toBlob();
  return blob;
};
