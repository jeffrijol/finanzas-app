1. Servicio de Excel - Mejorar el Parser

Tu ExcelParser es bueno, pero sugiero esta optimización para manejar mejor tu formato específico:

// En packages/backend/src/utils/excelParser.ts
private static findDataStartRow(data: any[][]): number {
  // Tu Excel tiene esta estructura:
  // Fila 0-3: Metadata (Número de cuenta, Empresa, etc.)
  // Fila 4: Vacía
  // Fila 5: Headers (FECHA CONTABLE, FECHA VALOR, etc.)
  // Fila 6+: Datos
  
  // Buscar "FECHA CONTABLE" o "FECHA VALOR"
  for (let i = 0; i < Math.min(10, data.length); i++) {
    const row = data[i];
    if (row && row.some(cell => 
      String(cell).toUpperCase().includes('FECHA CONTABLE') || 
      String(cell).toUpperCase().includes('FECHA VALOR')
    )) {
      return i;
    }
  }
  return 5; // Fallback a la fila 5 (basado en tu estructura)
}

// En packages/backend/src/utils/apiResponse.ts
export function successResponse<T>(data: T, message?: string) {
  return {
    success: true,
    data,
    message,
    timestamp: new Date().toISOString(),
  };
}

export function errorResponse(message: string, error?: any) {
  return {
    success: false,
    error: message,
    details: process.env.NODE_ENV === 'development' ? error : undefined,
    timestamp: new Date().toISOString(),
  };
}
