🔍 Diagnóstico del Problema: Gráficos que No Aparecen
Posibles Causas:

    renderToString captura SVG vacío (Recharts no terminó de renderizar)

    Formato incorrecto del SVG para @react-pdf/renderer

    Dimensiones o estilos que hacen el SVG invisible

    Error silencioso en la conversión a base64/data URL

🛠️ Solución Probable: Problemas Comunes y Cómo Solucionarlos
Solución 1: renderToString captura SVG vacío

Causa: Recharts necesita que el DOM esté montado completamente.
Solución: Usar setTimeout o requestAnimationFrame:
typescript

export const renderBarChartToSVG = async (data: any): Promise<string> => {
  return new Promise((resolve) => {
    // Dar tiempo a React/Recharts para renderizar completamente
    setTimeout(() => {
      const chartElement = (
        <div style={{ width: 600, height: 300 }}>
          <BarChart data={data} width={600} height={300}>
            {/* Quitar ResponsiveContainer para renderizado estático */}
            <CartesianGrid strokeDasharray="3 3" />
            <XAxis dataKey="name" />
            <YAxis />
            <Tooltip />
            <Legend />
            <Bar dataKey="ingresos" fill="#10b981" />
            <Bar dataKey="gastos" fill="#ef4444" />
          </BarChart>
        </div>
      );
      
      const htmlString = renderToString(chartElement);
      const svgMatch = htmlString.match(/<svg[\s\S]*?<\/svg>/);
      resolve(svgMatch ? svgMatch[0] : '');
    }, 100); // Pequeño delay
  });
};

Solución 2: SVG con estilos/propiedades inválidas

Causa: Recharts añade propiedades que React-PDF no soporta.
Solución: Limpiar el SVG:
typescript

const cleanSVG = (svgString: string): string => {
  return svgString
    // Remover atributos problemáticos
    .replace(/style="[^"]*"/g, '')
    .replace(/class="[^"]*"/g, '')
    .replace(/width="[^"]*"/g, 'width="600"')
    .replace(/height="[^"]*"/g, 'height="300"')
    // Asegurar viewBox
    .replace(/<svg/, '<svg viewBox="0 0 600 300"')
    // Remover scripts o elementos interactivos
    .replace(/<script[\s\S]*?<\/script>/g, '')
    .replace(/on\w+="[^"]*"/g, '');
};

Solución 3: Formato incorrecto para React-PDF

Causa: React-PDF necesita SVG como string XML puro, no como data URL.
Solución alternativa: Usar directamente el string SVG:
tsx

// En pdf-builder.tsx, en lugar de:
<Image src={`data:image/svg+xml;base64,${btoa(svgString)}`} />

// Intentar esto:
<svg
  width="600"
  height="300"
  viewBox="0 0 600 300"
  dangerouslySetInnerHTML={{ __html: svgString.replace(/<svg[\s\S]*?>|<\/svg>/g, '') }}
/>

🎯 Código Corregido: Versión Funcional de chart-renderer.tsx
tsx

import React from 'react';
import { renderToString } from 'react-dom/server';
import {
  BarChart, Bar, PieChart, Pie, Cell,
  XAxis, YAxis, CartesianGrid, Tooltip, Legend,
  ResponsiveContainer
} from 'recharts';

/**
 * Solución robusta para renderizar gráficos Recharts a SVG
 */
export const renderChartToSVG = async (
  type: 'bar' | 'pie',
  data: any,
  options: {
    width?: number;
    height?: number;
    colors?: string[];
  } = {}
): Promise<string> => {
  const { width = 600, height = 300, colors = ['#10b981', '#ef4444'] } = options;
  
  return new Promise((resolve) => {
    // Delay para asegurar renderizado completo
    setTimeout(() => {
      try {
        let chartElement: React.ReactElement;
        
        if (type === 'bar') {
          chartElement = (
            <div style={{ width, height }}>
              <ResponsiveContainer width="100%" height="100%">
                <BarChart
                  data={data}
                  margin={{ top: 20, right: 30, left: 20, bottom: 5 }}
                >
                  <CartesianGrid strokeDasharray="3 3" stroke="#e5e7eb" />
                  <XAxis 
                    dataKey="name" 
                    stroke="#6b7280"
                    fontSize={12}
                  />
                  <YAxis 
                    stroke="#6b7280"
                    fontSize={12}
                    tickFormatter={(value) => `€${value}`}
                  />
                  <Tooltip 
                    formatter={(value) => [`€${value}`, '']}
                    contentStyle={{
                      backgroundColor: '#1f2937',
                      border: 'none',
                      borderRadius: '6px'
                    }}
                  />
                  <Legend />
                  <Bar 
                    dataKey="ingresos" 
                    name="Ingresos" 
                    fill={colors[0]}
                    radius={[4, 4, 0, 0]}
                  />
                  <Bar 
                    dataKey="gastos" 
                    name="Gastos" 
                    fill={colors[1]}
                    radius={[4, 4, 0, 0]}
                  />
                </BarChart>
              </ResponsiveContainer>
            </div>
          );
        } else {
          // Gráfico de torta
          chartElement = (
            <div style={{ width, height }}>
              <ResponsiveContainer width="100%" height="100%">
                <PieChart>
                  <Pie
                    data={data}
                    cx="50%"
                    cy="50%"
                    labelLine={false}
                    outerRadius={80}
                    fill="#8884d8"
                    dataKey="value"
                    label={(entry) => `${entry.name}: ${entry.value}`}
                  >
                    {data.map((entry: any, index: number) => (
                      <Cell 
                        key={`cell-${index}`} 
                        fill={colors[index % colors.length]} 
                      />
                    ))}
                  </Pie>
                  <Tooltip formatter={(value) => `€${value}`} />
                  <Legend />
                </PieChart>
              </ResponsiveContainer>
            </div>
          );
        }
        
        const htmlString = renderToString(chartElement);
        const svgMatch = htmlString.match(/<svg[\s\S]*?<\/svg>/);
        
        if (!svgMatch) {
          console.error('No se encontró SVG en el renderizado');
          resolve('');
          return;
        }
        
        let svg = svgMatch[0];
        
        // Limpiar y normalizar el SVG
        svg = svg
          .replace(/<svg/, `<svg width="${width}" height="${height}"`)
          .replace(/style="[^"]*"/g, '')
          .replace(/class="[^"]*"/g, '')
          .replace(/\s+/g, ' ')
          .trim();
        
        console.log(`✅ SVG ${type} generado: ${svg.length} caracteres`);
        resolve(svg);
        
      } catch (error) {
        console.error(`Error renderizando gráfico ${type}:`, error);
        resolve('');
      }
    }, 50);
  });
};

const svgToPNG = async (svgString: string): Promise<string> => {
  const img = new Image();
  const canvas = document.createElement('canvas');
  const ctx = canvas.getContext('2d');
  
  return new Promise((resolve) => {
    img.onload = () => {
      canvas.width = img.width;
      canvas.height = img.height;
      ctx?.drawImage(img, 0, 0);
      resolve(canvas.toDataURL('image/png'));
    };
    img.src = `data:image/svg+xml;base64,${btoa(svgString)}`;
  });
};
